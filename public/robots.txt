# Personal whiteboard: opt-out of indexing and AI training.
# Disallow:/ for User-agent: * already covers everything that respects RFC 9309,
# but the explicit blocks below are for crawlers that pattern-match their own
# UA before falling back to *, and for documenting intent.
#
# Agent discovery files are Allow-listed so known-URL agents can fetch them
# without opening the app UI to crawlers. Start at /llms.txt.
# Social link-preview bots are allowed so shared links render og:title/og:image.

Disallow: /
Allow: /llms.txt
Allow: /auth.md
Allow: /for-agents.html
Allow: /openapi.json
Allow: /.well-known/api-catalog
Allow: /.well-known/ard.json
Allow: /.well-known/ai-catalog.json
Allow: /.well-known/agent-skills/
